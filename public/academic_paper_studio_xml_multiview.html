<paper> <meta> <title>Judul Makalah</title> <subtitle>Opsional</subtitle> <course>Mata Kuliah</course> <institution>STIESIA</institution> </meta> <content> <heading level="1">BAB I PENDAHULUAN</heading> <heading level="2">1.1 Latar Belakang</heading> <paragraph>Isi paragraf...</paragraph> <table caption="Tabel 1. Contoh" source="Data diolah, 2026"> <thead> <row><cell>Keterangan</cell><cell>2025</cell></row> </thead> <tbody> <row><cell>Pendapatan</cell><cell type="number">12000000</cell></row> </tbody> </table> </content> <references> <reference>Referensi 1...</reference> </references> </paper>
